4//! `netevd explain`: show what the daemon would do for a synthetic or
5//! recorded event, without running anything.
6//!
7//! The report is built from the same functions the executor uses
8//! (`hook_dir`, `InterfaceSelector`, `plan_directory`, `prepare_env`), so it
9//! cannot drift from real behaviour.
10

23/// Longest value printed for one variable.
24const SHOW_VALUE_MAX: usize = 120;
25/// Longest JSONL line `load_events` will parse.
26pub const MAX_EVENT_LINE: usize = 1 << 20;
27/// Most events `load_events` returns.
28pub const MAX_EVENTS: usize = 10_000;

30#[derive(Debug, Clone, Serialize, PartialEq)]
31pub struct ScriptLine {
32    pub name: String,
33    /// `run` or `skip: <reason>`
34    pub outcome: String,
35}

37#[derive(Debug, Clone, Serialize, PartialEq)]
38pub struct EnvLine {
39    pub name: String,
40    pub value: String,
41}

43#[derive(Debug, Clone, Serialize, PartialEq)]
44pub struct Explanation {
45    pub event: String,
46    pub link: String,
47    pub backend: String,
48    pub hook_dir: Option<String>,
49    pub interface_selected: bool,
50    pub interface_reason: String,
51    pub filter_decision: String,
52    pub debounce_ms: u64,
53    pub rate_limit: String,
54    pub scripts: Vec<ScriptLine>,
55    pub env: Vec<EnvLine>,
56    pub env_rejected: Vec<EnvLine>,
57    /// True when at least one script would be executed.
58    pub would_run: bool,
59    pub notes: Vec<String>,
60}

62fn shorten(v: &str) -> String {
63    let flat: String = v
64        .chars()
65        .map(|c| if c.is_control() { '?' } else { c })
66        .collect();
67    if flat.chars().count() > SHOW_VALUE_MAX {
68        let cut: String = flat.chars().take(SHOW_VALUE_MAX).collect();
69        format!("{cut}... ({} bytes)", v.len())
70    } else {
71        flat
72    }
73}

75fn interface_decision(sel: &InterfaceSelector, link: &str) -> (bool, String) {
76    use super::match_iface::glob_matches;
77    if let Some(p) = sel.exclude.iter().find(|p| glob_matches(p, link)) {
78        return (false, format!("excluded by pattern '{p}'"));
79    }
80    if sel.include.is_empty() {
81        return (true, "no include list; not excluded".into());
82    }
83    match sel.include.iter().find(|p| glob_matches(p, link)) {
84        Some(p) => (true, format!("matches include pattern '{p}'")),
85        None => (false, "matches no include pattern".into()),
86    }
87}

89fn filter_decision(config: &Config, ev: &HookEventV1) -> String {
90    if ev.backend == "netlink" {
91        return "not applied (filters only apply to backend events, not netlink hooks)".into();
92    }
93    let addresses: Vec<IpAddr> = ev.addresses.iter().filter_map(|a| a.parse().ok()).collect();
94    let ne = NetworkEvent {
95        interface: ev.link.clone(),
96        event_type: ev.event.clone(),
97        backend: ev.backend.clone(),
98        addresses,
99        has_gateway: ev.gateway.is_some(),
100        dns_servers: ev.dns.iter().filter_map(|a| a.parse().ok()).collect(),
101    };
102    for (i, f) in config.filters.iter().enumerate() {
103        if f.matches(&ne) {
104            match f.action {
105                FilterAction::Execute => return format!("execute (filter #{})", i + 1),
106                FilterAction::Ignore => return format!("ignore (filter #{})", i + 1),
107                FilterAction::Log => continue,
108            }
109        }
110    }
111    "execute (no filter matched; default)".into()
112}

114/// Build the explanation. `daemon_uid` and `daemon_env` are injected so the
115/// result is reproducible in tests.
116pub fn explain_event(
117    ev: &HookEventV1,
118    config: &Config,
119    config_root: &Path,
120    daemon_uid: u32,
121    daemon_env: &dyn Fn(&str) -> Option<String>,
122) -> Explanation {
123    let mut notes = Vec::new();
124    let sandbox_cfg = &config.hooks.sandbox;
125
126    let dir = hook_dir(config_root, &ev.event);
127    if dir.is_none() {
128        notes.push(format!(
129            "'{}' is not a known hook state; the daemon rejects it and runs nothing",
130            shorten(&ev.event)
131        ));
132    }
133
134    let selector = InterfaceSelector::from_lists(
135        config.monitoring.match_patterns.clone(),
136        config.monitoring.exclude.clone(),
137    )
138    .with_default_excludes();
139    let (selected, interface_reason) = if ev.backend == "netlink" {
140        interface_decision(&selector, &ev.link)
141    } else {
142        (
143            true,
144            "interface selector only applies to netlink hooks".to_string(),
145        )
146    };
147    if !selected {
148        notes.push("netlink watchers drop this event before dispatch".into());
149    }
150
151    let filter = filter_decision(config, ev);
152
153    let mut scripts = Vec::new();
154    if let Some(d) = &dir {
155        match plan_directory(d, sandbox_cfg, daemon_uid) {
156            Ok(plan) => {
157                for p in plan {
158                    scripts.push(ScriptLine {
159                        name: shorten(&p.name),
160                        outcome: match p.verdict {
161                            Verdict::Run => "run".into(),
162                            Verdict::Skip(why) => format!("skip: {why}"),
163                        },
164                    });
165                }
166                if scripts.is_empty() {
167                    notes.push(format!("{} is missing or empty", d.display()));
168                }
169            }
170            Err(e) => notes.push(format!("cannot read {}: {e}", d.display())),
171        }
172    }
173
174    let raw_env = ev.to_env();
175    let prepared = sandbox::prepare_env(&raw_env, sandbox_cfg, daemon_env);
176    let env = prepared
177        .vars
178        .iter()
179        .map(|(k, v)| EnvLine {
180            name: k.clone(),
181            value: shorten(v),
182        })
183        .collect();
184    let env_rejected = prepared
185        .rejected
186        .iter()
187        .map(|(k, why)| EnvLine {
188            name: shorten(k),
189            value: why.clone(),
190        })
191        .collect();
192
193    let rl = &config.hooks.rate_limit;
194    let rate_limit = if rl.enabled() {
195        format!(
196            "burst {} then {}/min per (link, event)",
197            rl.burst, rl.per_minute
198        )
199    } else {
200        "off".into()
201    };
202
203    let gate_open = selected && !filter.starts_with("ignore");
204    let would_run = gate_open && scripts.iter().any(|s| s.outcome == "run");
205
206    Explanation {
207        event: shorten(&ev.event),
208        link: shorten(&ev.link),
209        backend: shorten(&ev.backend),
210        hook_dir: dir.map(|d| d.display().to_string()),
211        interface_selected: selected,
212        interface_reason,
213        filter_decision: filter,
214        debounce_ms: config.hooks.debounce_ms,
215        rate_limit,
216        scripts,
217        env,
218        env_rejected,
219        would_run,
220        notes,
221    }
222}

224impl Explanation {
225    pub fn render_text(&self) -> String {
226        use std::fmt::Write;
227        let mut o = String::new();
228        let _ = writeln!(
229            o,
230            "Event     : {} on {} ({})",
231            self.event, self.link, self.backend
232        );
233        let _ = writeln!(
234            o,
235            "Hook dir  : {}",
236            self.hook_dir.as_deref().unwrap_or("(none: unknown state)")
237        );
238        let _ = writeln!(
239            o,
240            "Interface : {} ({})",
241            if self.interface_selected {
242                "selected"
243            } else {
244                "dropped"
245            },
246            self.interface_reason
247        );
248        let _ = writeln!(o, "Filters   : {}", self.filter_decision);
249        let _ = writeln!(o, "Debounce  : {} ms", self.debounce_ms);
250        let _ = writeln!(o, "Rate limit: {}", self.rate_limit);
251        let _ = writeln!(o, "Scripts   :");
252        if self.scripts.is_empty() {
253            let _ = writeln!(o, "  (none)");
254        }
255        for s in &self.scripts {
256            let _ = writeln!(o, "  {:<32} {}", s.name, s.outcome);
257        }
258        let _ = writeln!(o, "Environment ({} variables):", self.env.len());
259        for e in &self.env {
260            let _ = writeln!(o, "  {}={}", e.name, e.value);
261        }
262        if !self.env_rejected.is_empty() {
263            let _ = writeln!(o, "Rejected variables:");
264            for e in &self.env_rejected {
265                let _ = writeln!(o, "  {} ({})", e.name, e.value);
266            }
267        }
268        for n in &self.notes {
269            let _ = writeln!(o, "Note      : {n}");
270        }
271        let _ = writeln!(
272            o,
273            "Result    : {}",
274            if self.would_run {
275                "scripts WOULD run (nothing was executed)"
276            } else {
277                "no script would run"
278            }
279        );
280        o
281    }
282}

284/// Parse a JSON-lines file of `netevd.event.v1` events. Malformed, oversized
285/// or wrong-schema lines are reported, never fatal; the result is capped.
286pub fn load_events(text: &str) -> (Vec<HookEventV1>, Vec<String>) {
287    let mut events = Vec::new();
288    let mut errors = Vec::new();
289    for (i, line) in text.lines().enumerate() {
290        let n = i + 1;
291        if line.trim().is_empty() {
292            continue;
293        }
294        if events.len() >= MAX_EVENTS {
295            errors.push(format!("line {n}: stopped after {MAX_EVENTS} events"));
296            break;
297        }
298        if line.len() > MAX_EVENT_LINE {
299            errors.push(format!(
300                "line {n}: longer than {MAX_EVENT_LINE} bytes, skipped"
301            ));
302            continue;
303        }
304        match serde_json::from_str::<HookEventV1>(line) {
305            Ok(e) if e.schema == super::event::SCHEMA_V1 => events.push(e),
306            Ok(e) => errors.push(format!(
307                "line {n}: unsupported schema '{}'",
308                shorten(&e.schema)
309            )),
310            Err(e) => errors.push(format!("line {n}: {e}")),
311        }
312    }
313    (events, errors)
314}

316/// Build a synthetic event from CLI-style parts.
317pub fn synthetic_event(
318    event: &str,
319    link: &str,
320    index: u32,
321    backend: &str,
322    addresses: Vec<String>,
323) -> HookEventV1 {
324    HookEventV1::new(event, link, index, backend).with_addresses(addresses)

352    #[test]
353    fn lists_scripts_in_order_and_never_executes() {
354        let (t, cfg) = setup();
355        let ev = synthetic_event("routable", "eth0", 2, "netlink", vec!["10.0.0.5".into()]);
356        let x = explain_event(&ev, &cfg, t.path(), 0, &no_env);
357        let names: Vec<_> = x
358            .scripts
359            .iter()
360            .map(|s| (s.name.as_str(), s.outcome.as_str()))
361            .collect();
362        assert_eq!(
363            names,
364            [
365                ("01-a.sh", "run"),
366                ("02-b.sh", "run"),
367                ("03-off", "skip: not executable")
368            ]
369        );
370        assert!(x.would_run);
371        assert!(x.interface_selected);
372        assert!(!Path::new("/tmp/SHOULD_NEVER_EXIST_netevd").exists());
373        assert!(x.render_text().contains("nothing was executed"));
374    }

459    #[test]
460    fn load_events_survives_garbage() {
461        let good = synthetic_event("routable", "eth0", 2, "netlink", vec![])
462            .to_json()
463            .unwrap();
464        let wrong = good.replace("netevd.event.v1", "netevd.event.v9");
465        let huge = format!("{{\"x\":\"{}\"}}", "a".repeat(MAX_EVENT_LINE + 1));
466        let text = format!(
