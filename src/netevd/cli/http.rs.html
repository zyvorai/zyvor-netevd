4//! Minimal HTTP/1.0 client for the CLI to talk to the local REST API. HTTP/1.0
5//! makes the server close the connection after the body and never use chunked
6//! encoding, so a plain read-to-end is a complete parser. Plain `http://` only.
7
8use anyhow::{bail, Context, Result};
9use std::time::Duration;
10use tokio::io::{AsyncBufReadExt, AsyncReadExt, AsyncWriteExt, BufReader};
11use tokio::net::TcpStream;
12
13const TIMEOUT: Duration = Duration::from_secs(10);
14const MAX_BODY: usize = 16 * 1024 * 1024;
15
16/// `http://host:port[/]` or bare `host:port` to `host:port`.
17pub fn parse_endpoint(endpoint: &str) -> Result<String> {
18    let e = endpoint.trim();
19    if e.starts_with("https://") {
20        bail!("https endpoints are not supported by the CLI client; use an http:// address");
21    }
22    let e = e.strip_prefix("http://").unwrap_or(e).trim_end_matches('/');
23    if e.is_empty() || e.contains('/') || e.contains(' ') {
24        bail!("invalid endpoint '{endpoint}': expected http://host:port");
25    }
26    Ok(if e.contains(':') && !e.ends_with(']') {
27        e.to_string()
28    } else {
29        format!("{e}:80")
30    })
31}

33/// Percent-encode a query value.
34pub fn encode(v: &str) -> String {
35    let mut out = String::new();
36    for b in v.bytes() {
37        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'_' | b'.' | b'~') {
38            out.push(b as char);
39        } else {
40            out.push_str(&format!("%{b:02X}"));
41        }
42    }
43    out
44}

46fn request(method: &str, host: &str, path: &str, body: Option<&str>) -> String {
47    let host = host.to_string();
48    match body {
49        Some(b) => format!(
50            "{method} {path} HTTP/1.0\r\nHost: {host}\r\nAccept: application/json\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{b}",
51            b.len()
52        ),
53        None => format!(
54            "{method} {path} HTTP/1.0\r\nHost: {host}\r\nAccept: */*\r\nConnection: close\r\n\r\n"
55        ),
56    }
57}

59async fn connect(endpoint: &str) -> Result<(TcpStream, String)> {
60    let addr = parse_endpoint(endpoint)?;
61    let s = tokio::time::timeout(TIMEOUT, TcpStream::connect(&addr))
62        .await
63        .with_context(|| format!("timed out connecting to {addr}"))?
64        .with_context(|| {
65            format!(
66                "cannot connect to netevd at {addr} (is the daemon running with the API enabled?)"
67            )
68        })?;
69    Ok((s, addr))
70}

72fn split_response(raw: &[u8]) -> Result<(u16, &[u8])> {
73    let pos = raw
74        .windows(4)
75        .position(|w| w == b"\r\n\r\n")
76        .context("malformed HTTP response")?;
77    let head = std::str::from_utf8(&raw[..pos]).context("non-UTF-8 HTTP header")?;
78    let status = head
79        .lines()
80        .next()
81        .and_then(|l| l.split_whitespace().nth(1))
82        .and_then(|c| c.parse::<u16>().ok())
83        .context("malformed HTTP status line")?;
84    Ok((status, &raw[pos + 4..]))
85}

87async fn roundtrip(
88    endpoint: &str,
89    method: &str,
90    path: &str,
91    body: Option<&str>,
92) -> Result<Vec<u8>> {
93    let (mut s, addr) = connect(endpoint).await?;
94    s.write_all(request(method, &addr, path, body).as_bytes())
95        .await?;
96    let mut raw = Vec::new();
97    tokio::time::timeout(
98        TIMEOUT,
99        (&mut s).take(MAX_BODY as u64).read_to_end(&mut raw),
100    )
101    .await
102    .context("timed out waiting for the response")??;
103    let (status, body) = split_response(&raw)?;
104    if !(200..300).contains(&status) {
105        bail!(
106            "{method} {path} failed with HTTP {status}: {}",
107            String::from_utf8_lossy(body).trim()
108        );
109    }
110    Ok(body.to_vec())
111}

113/// GET `path` (with query) and parse the JSON body.
114pub async fn get_json<T: serde::de::DeserializeOwned>(endpoint: &str, path: &str) -> Result<T> {
115    let body = roundtrip(endpoint, "GET", path, None).await?;
116    serde_json::from_slice(&body).with_context(|| format!("unexpected response from {path}"))
117}
118
119/// POST a JSON body and parse the JSON response.
120pub async fn post_json<T: serde::de::DeserializeOwned>(
121    endpoint: &str,
122    path: &str,
123    body: &str,
124) -> Result<T> {
125    let body = roundtrip(endpoint, "POST", path, Some(body)).await?;
126    serde_json::from_slice(&body).with_context(|| format!("unexpected response from {path}"))
127}

129/// The payload of an SSE `data:` line, if it is one.
130pub fn sse_data(line: &str) -> Option<&str> {
131    line.strip_prefix("data:").map(str::trim_start)
132}
133
134/// GET an SSE stream and call `on_data` for each `data:` payload until the
135/// server closes the connection or `on_data` returns false.
136pub async fn stream_sse(
137    endpoint: &str,
138    path: &str,
139    mut on_data: impl FnMut(&str) -> bool,
140) -> Result<()> {
141    let (mut s, addr) = connect(endpoint).await?;
142    s.write_all(request("GET", &addr, path, None).as_bytes())
143        .await?;
144    let mut lines = BufReader::new(s).lines();
145    let status_line = lines
146        .next_line()
147        .await?
148        .context("connection closed before a response")?;
149    let status = status_line
150        .split_whitespace()
151        .nth(1)
152        .and_then(|c| c.parse::<u16>().ok())
153        .context("malformed HTTP status line")?;
154    if status != 200 {
155        bail!("GET {path} failed with HTTP {status}");
156    }
157    while let Some(line) = lines.next_line().await? {
158        if let Some(d) = sse_data(&line) {
159            if !on_data(d) {
160                break;
161            }
162        }
163    }
164    Ok(())
165}

167#[cfg(test)]
168mod tests {
169    use super::*;
170
171    #[test]
172    fn endpoint_forms() {
173        assert_eq!(
174            parse_endpoint("http://localhost:9090").unwrap(),
175            "localhost:9090"
176        );
177        assert_eq!(
178            parse_endpoint("http://localhost:9090/").unwrap(),
179            "localhost:9090"
180        );
181        assert_eq!(parse_endpoint("127.0.0.1:1").unwrap(), "127.0.0.1:1");
182        assert_eq!(parse_endpoint("http://example").unwrap(), "example:80");
183        assert_eq!(parse_endpoint("http://[::1]:9").unwrap(), "[::1]:9");
184        assert!(parse_endpoint("https://x:1").is_err());
185        assert!(parse_endpoint("http://x:1/path").is_err());
186        assert!(parse_endpoint("").is_err());
187    }
188
189    #[test]
190    fn query_values_are_encoded() {
191        assert_eq!(encode("eth0"), "eth0");
192        assert_eq!(encode("a b&c=d"), "a%20b%26c%3Dd");
193    }
194
195    #[test]
196    fn response_split_and_sse_lines() {
197        let (st, body) = split_response(b"HTTP/1.0 200 OK\r\nX: y\r\n\r\n{\"a\":1}").unwrap();
198        assert_eq!((st, body), (200, &b"{\"a\":1}"[..]));
199        assert!(split_response(b"garbage").is_err());
200        assert_eq!(sse_data("data: {\"x\":1}"), Some("{\"x\":1}"));
201        assert_eq!(sse_data("id: 4"), None);
202        assert_eq!(sse_data(": keep-alive"), None);
203    }
